      <button id="tabTrendBtn" onclick="switchTab('trend')" class="px-5 py-2 rounded-lg text-sm font-semibold transition-all duration-200 text-slate-600 hover:text-slate-900">📈 單元趨勢總覽</button>

      </div>
      <div class="grid grid-cols-1 lg:grid-cols-2 gap-6">
        <div class="bg-white p-5 rounded-xl border border-slate-200 shadow-sm flex flex-col items-center">
          <h3 class="text-sm font-bold text-slate-800 mb-4 self-start flex items-center"><span class="w-1.5 h-4 bg-indigo-600 rounded-full mr-2"></span>大單元出題比重 (章分布)</h3>
          <div class="w-full max-w-[320px] aspect-square flex items-center justify-center"><canvas id="pieChart"></canvas></div>
        </div>
        <div class="bg-white p-5 rounded-xl border border-slate-200 shadow-sm">
          <h3 class="text-sm font-bold text-slate-800 mb-4 flex items-center"><span class="w-1.5 h-4 bg-indigo-600 rounded-full mr-2"></span>細分主題題數分佈 (節分布)</h3>
          <div class="w-full h-[320px]"><canvas id="barChart"></canvas></div>
        </div>
      </div>
      <div class="space-y-4">
        <h3 class="text-base font-bold text-slate-900 flex items-center">該屆全卷試題清單與節點</h3>
        <div id="questionsListContainer" class="grid grid-cols-1 gap-4"></div>
      </div>
    </section>

    <section id="viewTrend" class="space-y-6 hidden">
      <div class="bg-white p-5 rounded-xl border border-slate-200 shadow-sm grid grid-cols-1 md:grid-cols-2 gap-4">
        <div>
          <label class="block text-xs font-bold text-slate-500 mb-1">第一層：大單元 (章)</label>
          <select id="selectUnit" onchange="onUnitChange()" class="w-full bg-slate-50 border rounded-lg p-2.5 text-sm"></select>
        </div>
        <div>
          <label class="block text-xs font-bold text-slate-500 mb-1">第二層：細分主題 (節)</label>
          <select id="selectTopic" onchange="renderTrendView()" class="w-full bg-slate-50 border rounded-lg p-2.5 text-sm"></select>
        </div>
      </div>
      <div class="bg-white p-5 rounded-xl border border-slate-200 shadow-sm">
        <h3 class="text-sm font-bold text-slate-800 mb-4 flex items-center"><span class="w-1.5 h-4 bg-emerald-500 rounded-full mr-2"></span>歷屆出題趨勢折線圖</h3>
        <div class="w-full h-[280px]"><canvas id="lineChart"></canvas></div>
      </div>
      <div class="space-y-4">
        <h3 class="text-base font-bold text-slate-900">歷屆相同單元試題縱向匯整</h3>

  <script>
    // 嚴格遵守不腦補原則：115 年前 5 題為已知，其餘設定為「待歸類」
    const examDatabase = [
      { year: "115", id: "Q01", unit: "二元一次聯立方程式", topic: "解二元一次聯立方程式", img: "images/115_Q01.png", nodes: ["加減消去法的代數移項與運算"] },
      { year: "115", id: "Q02", unit: "立體幾何圖形", topic: "柱體、錐體與空間概念", img: "images/115_Q02.png", nodes: ["正三角形內角性質", "三維角度辨析"] },
      { year: "115", id: "Q03", unit: "平方根與畢氏定理", topic: "根式的運算", img: "images/115_Q03.png", nodes: ["最簡根式的化簡"] },
      { year: "115", id: "Q04", unit: "統計與機率", topic: "機率", img: "images/115_Q04.png", nodes: ["古典機率的定義"] },
      { year: "115", id: "Q05", unit: "整數運算與科學記號", topic: "整數的乘除運算", img: "images/115_Q05.png", nodes: ["負數運算變號法則"] },
      // 全卷擴充：第 6 ~ 25 題選擇題，以及 Q26, Q27 非選題 (佔位結構)
      ...Array.from({length: 20}, (_, i) => ({ year: "115", id: `Q${(i+6).toString().padStart(2, '0')}`, unit: "待歸類", topic: "待歸類", img: `images/115_Q${(i+6).toString().padStart(2, '0')}.png`, nodes: ["待分析節點"] })),
      { year: "115", id: "Q26", unit: "待歸類", topic: "待歸類", img: "images/115_Q26.png", nodes: ["非選一待分析"] },
      { year: "115", id: "Q27", unit: "待歸類", topic: "待歸類", img: "images/115_Q27.png", nodes: ["非選二待分析"] }
    ];

    const syllabusMapping = {
      "二元一次聯立方程式": ["解二元一次聯立方程式"],
      "立體幾何圖形": ["柱體、錐體與空間概念"],
      "平方根與畢氏定理": ["根式的運算"],
      "統計與機率": ["機率"],
      "整數運算與科學記號": ["整數的乘除運算"],
      "待歸類": ["待歸類"] // 避免系統錯誤的預設類別
    };

    const yearsList = ["115"]; // 目前僅先載入 115，日後可擴充 ["115", "114", "113"]
    let currentActiveYear = "115";
    let pieChartInstance = null, barChartInstance = null, lineChartInstance = null;

    window.onload = () => { renderYearSelectors(); initSyllabusDropdowns(); loadYearView(currentActiveYear); };

    function switchTab(tab) {
      document.getElementById('viewYear').classList.toggle('hidden', tab !== 'year');
      document.getElementById('viewTrend').classList.toggle('hidden', tab !== 'trend');
      if (tab === 'year') loadYearView(currentActiveYear); else renderTrendView();
    }

    function renderYearSelectors() {
      document.getElementById('yearSelectorContainer').innerHTML = yearsList.map(y => 
        `<button onclick="changeYear('${y}')" class="px-4 py-2 text-xs font-bold rounded-lg border bg-indigo-600 text-white">${y} 學年度</button>`
      ).join('');
    }

    function changeYear(year) { currentActiveYear = year; loadYearView(year); }

    function loadYearView(year) {
      const data = examDatabase.filter(d => d.year === year);
      const unitCounts = {}, topicCounts = {};
      data.forEach(d => { if(d.unit !== "待歸類") { unitCounts[d.unit] = (unitCounts[d.unit] || 0) + 1; topicCounts[d.topic] = (topicCounts[d.topic] || 0) + 1; }});
      
      renderPieChart(Object.keys(unitCounts), Object.values(unitCounts));
      renderBarChart(Object.keys(topicCounts), Object.values(topicCounts));
      
      document.getElementById('questionsListContainer').innerHTML = data.map(q => `
        <div class="bg-white rounded-xl border shadow-sm p-5 flex flex-col space-y-4">
          <div class="flex flex-wrap items-center gap-2">
            <span class="bg-indigo-50 text-indigo-700 text-xs font-bold px-2 py-1 border border-indigo-200">第 ${q.id.replace('Q','')} 題</span>
            <span class="${q.unit === '待歸類' ? 'bg-slate-300' : 'bg-slate-800'} text-white text-xs px-2 py-0.5 rounded">${q.unit}</span>
            <span class="${q.topic === '待歸類' ? 'bg-slate-200 text-slate-500' : 'bg-slate-500 text-white'} text-xs px-2 py-0.5 rounded">${q.topic}</span>
          </div>
          <div class="bg-slate-50 rounded-lg border p-4 flex justify-center items-center min-h-[100px]">
            <img src="${q.img}" alt="尚未產生截圖: ${q.img}" onerror="this.style.display='none'; this.nextElementSibling.style.display='block';" class="max-h-[220px] object-contain" />
            <p class="text-xs text-slate-400 hidden">圖片尚未匯入: ${q.img}</p>
          </div>
        </div>
      `).join('');
    }

    function initSyllabusDropdowns() {
      const uSelect = document.getElementById('selectUnit');
      uSelect.innerHTML = Object.keys(syllabusMapping).filter(u => u !== "待歸類").map(u => `<option value="${u}">${u}</option>`).join('');
      onUnitChange();
    }

    function onUnitChange() {
      const topic = syllabusMapping[document.getElementById('selectUnit').value] || [];
      document.getElementById('selectTopic').innerHTML = topic.map(t => `<option value="${t}">${t}</option>`).join('');
      renderTrendView();
    }

    function renderTrendView() {
      const t = document.getElementById('selectTopic').value;
      const trendData = yearsList.map(y => examDatabase.filter(d => d.year === y && d.topic === t).length).reverse();
      renderLineChart(["115年"], trendData, t); // 擴充年份時請同步修改標籤
      
      const qData = examDatabase.filter(d => d.topic === t);
      document.getElementById('trendQuestionsContainer').innerHTML = qData.length ? qData.map(q => `
        <div class="bg-white rounded-xl border p-4 flex flex-col space-y-3">
          <span class="text-xs font-bold text-emerald-700">${q.year}年 · 第 ${q.id.replace('Q','')} 題</span>
          <img src="${q.img}" onerror="this.style.display='none'; this.nextElementSibling.style.display='block';" class="max-h-[160px] object-contain" />
          <p class="text-xs text-slate-400 hidden">缺圖: ${q.img}</p>
        </div>`).join('') : '<p class="text-sm text-slate-400">暫無數據</p>';
    }

    function renderPieChart(lbls, data) {
      if(pieChartInstance) pieChartInstance.destroy();
      pieChartInstance = new Chart(document.getElementById('pieChart').getContext('2d'), { type: 'pie', data: { labels: lbls, datasets: [{ data: data, backgroundColor: ['#4f46e5', '#10b981', '#f59e0b', '#ef4444', '#8b5cf6'] }] } });
    }
    function renderBarChart(lbls, data) {
      if(barChartInstance) barChartInstance.destroy();
      barChartInstance = new Chart(document.getElementById('barChart').getContext('2d'), { type: 'bar', data: { labels: lbls, datasets: [{ label: '題數', data: data, backgroundColor: '#6366f1' }] }, options: { scales: { y: { beginAtZero: true, ticks: { stepSize: 1 } } } } });
    }
    function renderLineChart(lbls, data, title) {
      if(lineChartInstance) lineChartInstance.destroy();
      lineChartInstance = new Chart(document.getElementById('lineChart').getContext('2d'), { type: 'line', data: { labels: lbls, datasets: [{ label: title, data: data, borderColor: '#10b981', fill: true, tension: 0.1 }] }, options: { scales: { y: { beginAtZero: true, ticks: { stepSize: 1 } } } } });
    }
  </script>
